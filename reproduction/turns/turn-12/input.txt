import com.pico.spatial.ui.platform.containers.SpatialNavigator

object MuseumRoomStage {
    const val ID = "luna-museum-room"

    @Composable
    fun Content() {
        val navigator = LocalSpatialNavigator.current

        Column(modifier = Modifier.padding(20.dp)) {
            Text("Luna Museum room")
            Text("18 exhibits on plinths")
            Spacer(modifier = Modifier.height(8.dp))
            SpatialView(
                update = { _, _ -> },
                initial = { content, _ ->
                    content.addEntity(createRoomScene())
                },
            )
            Spacer(modifier = Modifier.height(8.dp))
            Button(onClick = { kotlinx.coroutines.GlobalScope.launch { close(navigator) } }) {
                Text("Leave museum room")
            }
        }
    }

    suspend fun open(navigator: SpatialNavigator) {
        MuseumRoomStore.clearMessage()
        when (val result = navigator.openStage(ID)) {
            OpenStageResult.Allowed -> Unit
            OpenStageResult.NotAllowed -> {
                MuseumRoomStore.showMessage("The museum room cannot open on this device right now.")
            }
            is OpenStageResult.Error -> {
                MuseumRoomStore.showMessage("The museum room could not open: ${result.reason}")
            }
        }
    }

    suspend fun close(navigator: SpatialNavigator) {
        navigator.closeStage()
    }
}

private fun createRoomScene(): Entity {
    val room = Entity()
    MuseumObject.values().forEachIndexed { index, _ ->
        val exhibit = Entity()
        exhibit.components[TransformComponent::class.java] = TransformComponent(
            position = roomPosition(index),
            rotation = EulerAngles(0f, 0f, 0f),
            scale = Vector3(1f, 1f, 1f),
        )
        exhibit.addChild(model(
            MeshResource.createBox(Vector3(0.8f, 0.35f, 0.8f), 0.06f),
            Color4(0.18f, 0.2f, 0.25f, 1f),
            Vector3(0f, -0.3f, 0f),
        ))
        exhibit.addChild(model(
            exhibitMesh(index),
            exhibitColor(index),
            Vector3(0f, 0.15f, 0f),
        ))
        room.addChild(exhibit)
    }
    return room
}

private fun roomPosition(index: Int): Vector3 {
    val column = index % 6
    val row = index / 6
    return Vector3((column - 2.5f) * 1.25f, 0f, (row - 1f) * 1.25f)
}

private fun exhibitMesh(index: Int): MeshResource =
    if (index % 2 == 0) {
        MeshResource.createBox(Vector3(0.55f, 0.55f, 0.55f), 0.08f)
    } else {
        MeshResource.createCylinder(0.28f, 0.6f)
    }

private fun exhibitColor(index: Int): Color4 = Color4(
    0.25f + (index % 4) * 0.12f,
    0.4f + (index % 3) * 0.12f,
    0.7f,
    1f,
)

private fun model(mesh: MeshResource, color: Color4, position: Vector3): ModelEntity {
    val material = UnlitMaterial.create()
    material.setBaseColor(color)
    return ModelEntity(mesh, material).also {
        it.components[TransformComponent::class.java] = TransformComponent(
            position = position,
            rotation = EulerAngles(0f, 0f, 0f),
            scale = Vector3(1f, 1f, 1f),
        )
    }
}

object MuseumRoomStore {
    private val messageState = mutableStateOf<String?>(null)

    val message: String?
        @Synchronized get() = messageState.value

    @Synchronized
    fun showMessage(value: String) {
        messageState.value = value
    }

    @Synchronized
    fun clearMessage() {
        messageState.value = null
    }
}

        val scope = rememberCoroutineScope()
        val selection = MuseumSelectionStore.current
        val story = MuseumStoryStore.current
        val roomMessage = MuseumRoomStore.message

            MuseumStoryStore.select(item)
        }

            modifier = Modifier

                .verticalScroll(rememberScrollState()),

            Text(selection.selected.description)
            Spacer(modifier = Modifier.height(8.dp))
            Text("Quarter turns: ${selection.turnQuarterTurns}")
            Text("Story ${story.stepIndex + 1} of ${story.story.steps.size}: ${story.stepText}")
            roomMessage?.let {
                Spacer(modifier = Modifier.height(8.dp))
                Text(it)
            }
            Spacer(modifier = Modifier.height(16.dp))
            Text("Catalogue")
            Spacer(modifier = Modifier.height(6.dp))

            MuseumObject.values().forEachIndexed { index, item ->
                Button(
                    onClick = { select(item) },
                    modifier = Modifier.fillMaxWidth(),
                ) {
                    Text("${index + 1}. ${item.title}${if (item.isPrimitivePlaceholder) " (primitive placeholder)" else ""}")
                }
                Spacer(modifier = Modifier.height(5.dp))
            }

            Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                Button(onClick = {
                    MuseumStoryStore.update(if (story.isPlaying) story.pause() else story.play())
                }) { Text(if (story.isPlaying) "Pause story" else "Play story") }
                Button(onClick = { MuseumStoryStore.update(story.restart()) }) { Text("Restart story") }
            }

            Spacer(modifier = Modifier.height(8.dp))
            Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                Button(onClick = { MuseumSelectionStore.update(selection.turn()) }) { Text("Turn") }
                Button(onClick = {
                    val resetSelection = MuseumSelectionStore.current.reset()
                    MuseumSelectionStore.update(resetSelection)
                    MuseumStoryStore.reset(resetSelection.selected)
                }) { Text("Reset") }
                Button(onClick = { MuseumStoryStore.update(story.toggleReducedMotion()) }) {
                    Text("Reduced motion: ${if (story.reducedMotion) "On" else "Off"}")
                }
            }

            Spacer(modifier = Modifier.height(8.dp))
            Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {

                    scope.launch { MuseumRoomStage.open(navigator) }
                }) { Text("Enter museum room") }
                Button(onClick = { navigator.closeWindowContainer() }) { Text("Close") }
            }
        }
    }

    FIDGET_TOY("Fidget toy", "Primitive placeholder: a simple rounded exhibit.", true),
    FACE_MASK("Face mask", "Primitive placeholder: a simple thin-box exhibit.", true),
    SMARTPHONE("Smartphone", "Primitive placeholder: a simple thin-box exhibit.", true),
    EARBUDS("Earbuds", "Primitive placeholder: a pair-like simple exhibit.", true),
    SMARTWATCH("Smartwatch", "Primitive placeholder: a simple rounded exhibit.", true),
    CRYPTO_WALLET("Crypto wallet", "Primitive placeholder: a simple wallet-shaped exhibit.", true),
    REMOTE_CONTROL("Remote control", "Primitive placeholder: a simple long-box exhibit.", true),
    ROUTER("Router", "Primitive placeholder: a simple box-shaped exhibit.", true),
    DOORBELL("Doorbell", "Primitive placeholder: a simple wall-box exhibit.", true),
    SCOOTER("Scooter", "Primitive placeholder: a simple bar-and-box exhibit.", true),
    GAS_CAN("Gas can", "Primitive placeholder: a simple can-shaped exhibit.", true),
    ROCKET("Rocket", "Primitive placeholder: a simple cone-and-box exhibit.", true),
}

data class MuseumSelection(
    val selected: MuseumObject = MuseumObject.HANDLED_CUP,
    val turnQuarterTurns: Int = 0,
) {
    fun select(item: MuseumObject): MuseumSelection = copy(selected = item, turnQuarterTurns = 0)
    fun turn(): MuseumSelection = copy(turnQuarterTurns = (turnQuarterTurns + 1) % 4)
    fun reset(): MuseumSelection = copy(turnQuarterTurns = 0)
}
