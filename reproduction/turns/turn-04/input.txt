fun HomeVolume() {
    var selection by remember { mutableStateOf(MuseumSelection()) }
    var museumScene by remember { mutableStateOf<MuseumScene?>(null) }

    Column(modifier = Modifier.padding(16.dp)) {
        Text("Luna Museum")
        Spacer(modifier = Modifier.height(8.dp))
        Text("Select an exhibit to inspect it in 3D.")
        Spacer(modifier = Modifier.height(12.dp))

        SpatialView(
            modifier = Modifier
                .fillMaxWidth()
                .size(260.dp)
                .clip(RoundedCornerShape(18.dp)),
            update = { _, _ ->
                museumScene?.apply(selection)
            },
            initial = { content, _ ->
                val scene = createMuseumScene()
                museumScene = scene
                content.addEntity(scene.root)
                scene.apply(selection)
            },
        )

        Spacer(modifier = Modifier.height(12.dp))
        Text(selection.selected.title)
        Text(selection.selected.description)
        Spacer(modifier = Modifier.height(10.dp))

        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.spacedBy(8.dp),

            Button(onClick = { selection = selection.select(MuseumObject.HANDLED_CUP) }) {
                Text("Cup")

private class MuseumScene(
    val root: Entity,
    private val objects: Map<MuseumObject, Entity>,
) {
    fun apply(selection: MuseumSelection) {
        objects.forEach { (museumObject, entity) ->
            entity.enabled = museumObject == selection.selected
            if (museumObject == selection.selected) {
                val transform = entity.components[TransformComponent::class.java] ?: return@forEach
                transform.eulerAngles = EulerAngles(
                    0f,
                    selection.turnQuarterTurns * 90f,
                    0f,
                )
            }
        }
    }
}

private fun createMuseumScene(): MuseumScene {
    val root = Entity()
    val cup = createCup()
    val vacuum = createRobotVacuum()
    val speaker = createSmartSpeaker()

    root.addChild(cup)
    root.addChild(vacuum)
    root.addChild(speaker)

    return MuseumScene(
        root = root,
        objects = mapOf(
            MuseumObject.HANDLED_CUP to cup,
            MuseumObject.ROBOT_VACUUM to vacuum,
            MuseumObject.SMART_SPEAKER to speaker,
        ),
    )
}

package dev.local.lunamuseum.reproduction.content

enum class MuseumObject(
    val title: String,
    val description: String,
) {
    HANDLED_CUP(
        title = "Handled cup",
        description = "A simple ceramic cup with a sturdy handle.",
    ),
    ROBOT_VACUUM(
        title = "Robot vacuum",
        description = "A compact robot vacuum that quietly patrols the floor.",
    ),
    SMART_SPEAKER(
        title = "Smart speaker",
        description = "A small connected speaker ready to answer a question.",
    ),
}

data class MuseumSelection(
    val selected: MuseumObject = MuseumObject.HANDLED_CUP,
    val turnQuarterTurns: Int = 0,
) {
    fun select(item: MuseumObject): MuseumSelection = copy(
        selected = item,
        turnQuarterTurns = 0,
    )

    fun turn(): MuseumSelection = copy(
        turnQuarterTurns = (turnQuarterTurns + 1) % 4,
    )

    fun reset(): MuseumSelection = copy(turnQuarterTurns = 0)
}

package dev.local.lunamuseum.reproduction.content

import org.junit.Assert.assertEquals
import org.junit.Test

class MuseumSelectionTest {
    @Test
    fun selectingAnObjectUpdatesTheDescriptionAndResetsTurn() {
        val turned = MuseumSelection().turn().turn()
        val selected = turned.select(MuseumObject.SMART_SPEAKER)

        assertEquals(MuseumObject.SMART_SPEAKER, selected.selected)
        assertEquals("Smart speaker", selected.selected.title)
        assertEquals(0, selected.turnQuarterTurns)
    }

    @Test
    fun resetReturnsTheSelectedObjectToItsOriginalRotation() {
        val reset = MuseumSelection(
            selected = MuseumObject.ROBOT_VACUUM,
            turnQuarterTurns = 3,
        ).reset()

        assertEquals(MuseumObject.ROBOT_VACUUM, reset.selected)
        assertEquals(0, reset.turnQuarterTurns)
    }

    @Test
    fun turningWrapsAfterFourQuarterTurns() {
        val result = MuseumSelection()
            .turn()
            .turn()
            .turn()
            .turn()

        assertEquals(0, result.turnQuarterTurns)
    }
}
