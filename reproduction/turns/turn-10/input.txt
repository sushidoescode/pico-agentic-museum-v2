private class MuseumScene(
    val root: Entity,
    private val objects: Map<MuseumObject, Entity>,
) {
    fun apply(selection: MuseumSelection, story: MuseumStoryState) {
        objects.forEach { (museumObject, entity) ->
            entity.enabled = museumObject == selection.selected
            if (museumObject == selection.selected) {
                val transform = entity.components[TransformComponent::class.java] ?: return@forEach
                val quarterTurn = selection.turnQuarterTurns * 90f
                val motion = museumStoryMotion(
                    objectType = museumObject,
                    stepIndex = story.stepIndex,
                    reducedMotion = story.reducedMotion,
                )
                transform.position = Vector3(
                    motion.positionX,
                    motion.positionY,
                    motion.positionZ,
                )
                transform.eulerAngles = EulerAngles(
                    motion.pitch,
                    quarterTurn + motion.yaw,
                    motion.roll,
                )
            }
        }
    }
}

}

package dev.local.lunamuseum.reproduction.content

import androidx.compose.runtime.mutableStateOf

class MuseumStory(
    val objectType: MuseumObject,
    val steps: List<String>,
) {
    val title: String
        get() = objectType.title
}

object MuseumStories {
    fun forObject(objectType: MuseumObject): MuseumStory = MuseumStory(
        objectType = objectType,
        steps = when (objectType) {
            MuseumObject.HANDLED_CUP -> listOf(
                "The cup waits on the table.",
                "A warm drink is poured in.",
                "The handle makes it easy to lift.",
            )
            MuseumObject.ROBOT_VACUUM -> listOf(
                "The robot vacuum starts at home.",
                "It rolls out to patrol the floor.",
                "It returns when the room is clean.",
            )
            MuseumObject.SMART_SPEAKER -> listOf(
                "The speaker is listening quietly.",
                "A question wakes its glowing light.",
                "It answers and settles back down.",
            )
        },
    )
}

data class MuseumStoryState(
    val selected: MuseumObject = MuseumObject.HANDLED_CUP,
    val stepIndex: Int = 0,
    val isPlaying: Boolean = false,
    val reducedMotion: Boolean = false,
) {
    val story: MuseumStory
        get() = MuseumStories.forObject(selected)

    val stepText: String
        get() = story.steps[stepIndex]

    fun select(objectType: MuseumObject): MuseumStoryState = copy(
        selected = objectType,
        stepIndex = 0,
        isPlaying = false,
    )

    fun play(): MuseumStoryState = if (stepIndex == story.steps.lastIndex) {
        copy(stepIndex = 0, isPlaying = true)
    } else {
        copy(isPlaying = true)
    }

    fun pause(): MuseumStoryState = copy(isPlaying = false)

    fun restart(): MuseumStoryState = copy(stepIndex = 0, isPlaying = true)

    fun reset(objectType: MuseumObject = selected): MuseumStoryState = copy(
        selected = objectType,
        stepIndex = 0,
        isPlaying = false,
    )

    fun advance(): MuseumStoryState = if (stepIndex < story.steps.lastIndex) {
        copy(stepIndex = stepIndex + 1)
    } else {
        copy(isPlaying = false)
    }

    fun toggleReducedMotion(): MuseumStoryState = copy(
        reducedMotion = !reducedMotion,
    )
}

object MuseumStoryStore {
    private val state = mutableStateOf(MuseumStoryState())

    val current: MuseumStoryState
        @Synchronized get() = state.value

    @Synchronized
    fun update(next: MuseumStoryState) {
        state.value = next
    }

    @Synchronized
    fun select(objectType: MuseumObject) {
        state.value = state.value.select(objectType)
    }

    @Synchronized
    fun reset(objectType: MuseumObject = state.value.selected) {
        state.value = state.value.reset(objectType)
    }
}

data class MuseumStoryMotionValues(
    val positionX: Float = 0f,
    val positionY: Float = 0f,
    val positionZ: Float = 0f,
    val pitch: Float = 0f,
    val yaw: Float = 0f,
    val roll: Float = 0f,
)

fun museumStoryMotion(
    objectType: MuseumObject,
    stepIndex: Int,
    reducedMotion: Boolean,
): MuseumStoryMotionValues {
    if (reducedMotion) return MuseumStoryMotionValues()

    return when (objectType) {
        MuseumObject.HANDLED_CUP -> when (stepIndex) {
            1 -> MuseumStoryMotionValues(positionY = 0.12f, pitch = -8f)
            2 -> MuseumStoryMotionValues(roll = 12f)
            else -> MuseumStoryMotionValues()
        }
        MuseumObject.ROBOT_VACUUM -> when (stepIndex) {
            1 -> MuseumStoryMotionValues(positionX = 0.16f)
            2 -> MuseumStoryMotionValues(positionX = -0.16f, yaw = 18f)
            else -> MuseumStoryMotionValues()
        }
        MuseumObject.SMART_SPEAKER -> when (stepIndex) {
            1 -> MuseumStoryMotionValues(positionY = 0.08f, pitch = -6f)
            2 -> MuseumStoryMotionValues(positionY = 0.03f, yaw = 10f)
            else -> MuseumStoryMotionValues()
        }
    }
}

        MuseumStoryStore.update(
            MuseumStoryState(
                selected = MuseumObject.ROBOT_VACUUM,
                stepIndex = 2,
                isPlaying = true,
                reducedMotion = true,
            ),
        )

        MuseumStoryStore.reset(MuseumObject.ROBOT_VACUUM)

        assertEquals(MuseumObject.ROBOT_VACUUM, MuseumStoryStore.current.selected)
        assertEquals(0, MuseumStoryStore.current.stepIndex)
        assertEquals("The robot vacuum starts at home.", MuseumStoryStore.current.stepText)
        assertFalse(MuseumStoryStore.current.isPlaying)
        assertTrue(MuseumStoryStore.current.reducedMotion)
    }
